<script>
 function selectPayment(type) {
 const cards = {
 vipps: document.getElementById('vipps-card'),
 card: document.getElementById('card-card'),
 cash: document.getElementById('cash-card')
 };
 const indicators = {
 vipps: document.getElementById('vipps-indicator'),
 card: document.getElementById('card-indicator'),
 cash: document.getElementById('cash-indicator')
 };
 const ctaLabel = document.getElementById('cta-label');

 // Reset all
 Object.keys(cards).forEach(key => {
 cards[key].classList.remove('bg-surface-container-lowest', 'shadow-md');
 cards[key].classList.add('bg-surface-container-low', 'shadow-sm');
 indicators[key].className = 'w-6 h-6 rounded-full bg-surface-container-highest flex items-center justify-center shrink-0 text-transparent';
 });

 // Activate selected
 cards[type].classList.remove('bg-surface-container-low', 'shadow-sm');
 cards[type].classList.add('bg-surface-container-lowest', 'shadow-md');
 indicators[type].className = 'w-6 h-6 rounded-full bg-secondary flex items-center justify-center shrink-0 text-on-secondary shadow-sm';

 if (type === 'vipps') {
 ctaLabel.textContent = 'Book Ride with Vipps';
 } else if (type === 'card') {
 ctaLabel.textContent = 'Book Ride with Card';
 } else if (type === 'cash') {
 ctaLabel.textContent = 'Confirm Cash Booking';
 }
 }
</script>
