<script>
 (function() {
 const inputs = document.querySelectorAll('.otp-field');
 const verifyBtn = document.getElementById('verify-button');
 const countdownEl = document.getElementById('countdown');
 const resendWrapper = document.getElementById('resend-wrapper');
 const resendBtn = document.getElementById('resend-btn');

 inputs.forEach((input, index) => {
 input.addEventListener('input', (e) => {
 const val = e.target.value;
 if (val.length === 1 && index < inputs.length - 1) {
 inputs[index + 1].focus();
 }
 highlightActive();
 });

 input.addEventListener('keydown', (e) => {
 if (e.key === 'Backspace' && !input.value && index > 0) {
 inputs[index - 1].focus();
 }
 });

 input.addEventListener('focus', () => {
 highlightActive();
 });
 });

 function highlightActive() {
 inputs.forEach(inp => {
 if (document.activeElement === inp) {
 inp.classList.remove('bg-surface-container-lowest');
 inp.classList.add('bg-secondary-container', 'text-on-secondary-container');
 } else {
 inp.classList.remove('bg-secondary-container', 'text-on-secondary-container');
 inp.classList.add('bg-surface-container-lowest');
 }
 });
 }

 let secondsLeft = 42;
 const timer = setInterval(() => {
 secondsLeft--;
 if (secondsLeft <= 0) {
 clearInterval(timer);
 resendWrapper.classList.add('hidden');
 resendBtn.classList.remove('hidden');
 } else {
 const formatted = secondsLeft < 10 ? '0' + secondsLeft : secondsLeft;
 countdownEl.textContent = '00:' + formatted;
 }
 }, 1000);

 resendBtn.addEventListener('click', () => {
 secondsLeft = 45;
 resendBtn.classList.add('hidden');
 resendWrapper.classList.remove('hidden');
 countdownEl.textContent = '00:45';
 });
 })();
 </script>
