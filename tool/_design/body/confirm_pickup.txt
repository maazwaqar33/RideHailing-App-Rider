<script>
 (() => {
 const noteBtn = document.getElementById('noteToggleBtn');
 const noteInputContainer = document.getElementById('noteInputContainer');
 const noteCaret = document.getElementById('noteCaret');
 const saveNoteBtn = document.getElementById('saveNoteBtn');
 const noteInput = document.getElementById('driverNoteInput');
 const noteButtonText = document.getElementById('noteButtonText');
 const recenterBtn = document.getElementById('recenterBtn');
 const trafficBtn = document.getElementById('trafficBtn');
 const tactilePin = document.getElementById('tactilePin');
 const confirmBtn = document.getElementById('confirmPickupBtn');

 if (noteBtn && noteInputContainer) {
 noteBtn.addEventListener('click', () => {
 const isHidden = noteInputContainer.classList.contains('hidden');
 if (isHidden) {
 noteInputContainer.classList.remove('hidden');
 noteCaret.classList.add('rotate-90');
 noteInput.focus();
 } else {
 noteInputContainer.classList.add('hidden');
 noteCaret.classList.remove('rotate-90');
 }
 });
 }

 if (saveNoteBtn && noteInput && noteButtonText) {
 saveNoteBtn.addEventListener('click', () => {
 const value = noteInput.value.trim();
 if (value.length > 0) {
 noteButtonText.textContent = `Note: "${value}"`;
 noteButtonText.classList.add('text-primary', 'font-semibold');
 } else {
 noteButtonText.textContent = 'Add pickup note for driver';
 noteButtonText.classList.remove('text-primary', 'font-semibold');
 }
 noteInputContainer.classList.add('hidden');
 noteCaret.classList.remove('rotate-90');
 });
 }

 if (recenterBtn && tactilePin) {
 recenterBtn.addEventListener('click', () => {
 tactilePin.classList.add('-translate-y-3');
 setTimeout(() => {
 tactilePin.classList.remove('-translate-y-3');
 }, 250);
 });
 }

 if (trafficBtn) {
 trafficBtn.addEventListener('click', () => {
 trafficBtn.classList.toggle('bg-secondary-container');
 });
 }

 if (confirmBtn) {
 confirmBtn.addEventListener('click', () => {
 confirmBtn.innerHTML = '<span class="material-symbols-outlined animate-spin text-[20px]">progress_activity</span>
<span class="font-title-sm text-title-sm">Connecting driver...</span>';
 setTimeout(() => {
 confirmBtn.innerHTML = '<span class="material-symbols-outlined text-[20px]">check_circle</span>
<span class="font-title-sm text-title-sm">Driver Assigned!</span>';
 confirmBtn.classList.add('bg-secondary');
 }, 1200);
 });
 }
 })();
</script>
