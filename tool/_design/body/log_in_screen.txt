<script>
 // Password Visibility Toggle
 const toggleBtn = document.getElementById('toggle-password');
 const passwordInput = document.getElementById('password');
 const eyeIcon = document.getElementById('eye-icon');

 if (toggleBtn && passwordInput && eyeIcon) {
 toggleBtn.addEventListener('click', () => {
 const isPassword = passwordInput.getAttribute('type') === 'password';
 passwordInput.setAttribute('type', isPassword ? 'text' : 'password');
 eyeIcon.textContent = isPassword ? 'visibility_off' : 'visibility';
 });
 }

 // Interactive Biometric Delight Trigger
 const triggerBiometricDelight = (button) => {
 const originalText = button.innerHTML;
 button.disabled = true;
 button.innerHTML = '<span class="material-symbols-outlined animate-spin text-[20px]">progress_activity</span>';
 
 setTimeout(() => {
 button.innerHTML = '<span class="material-symbols-outlined text-secondary text-[22px]">verified</span>';
 setTimeout(() => {
 button.innerHTML = originalText;
 button.disabled = false;
 }, 1400);
 }, 700);
 };

 const bioQuickBtn = document.getElementById('bio-quick-btn');
 if (bioQuickBtn) {
 bioQuickBtn.addEventListener('click', () => triggerBiometricDelight(bioQuickBtn));
 }

 const bioIconBtn = document.getElementById('biometric-icon-btn');
 if (bioIconBtn) {
 bioIconBtn.addEventListener('click', () => triggerBiometricDelight(bioIconBtn));
 }

 // Form Submission Feedback
 function triggerAuthFeedback() {
 const btn = document.getElementById('login-submit-btn');
 const textSpan = document.getElementById('btn-text');
 const iconSpan = document.getElementById('btn-icon');
 
 if (!btn || !textSpan || !iconSpan) return;

 btn.disabled = true;
 textSpan.textContent = 'Verifying...';
 iconSpan.textContent = 'progress_activity';
 iconSpan.classList.add('animate-spin');

 setTimeout(() => {
 textSpan.textContent = 'Welcome back!';
 iconSpan.classList.remove('animate-spin');
 iconSpan.textContent = 'check';
 btn.classList.remove('bg-primary-container');
 btn.classList.add('bg-secondary');
 
 setTimeout(() => {
 btn.disabled = false;
 textSpan.textContent = 'Log in';
 iconSpan.textContent = 'arrow_forward';
 btn.classList.remove('bg-secondary');
 btn.classList.add('bg-primary-container');
 }, 1500);
 }, 900);
 }
</script>
