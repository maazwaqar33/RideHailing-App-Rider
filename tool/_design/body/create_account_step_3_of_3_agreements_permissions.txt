<script>
 (function() {
 const biometricsBtn = document.getElementById('toggle-biometrics');
 const knob = document.getElementById('toggle-knob');
 let isBioActive = true;

 if (biometricsBtn && knob) {
 biometricsBtn.addEventListener('click', function() {
 isBioActive = !isBioActive;
 biometricsBtn.setAttribute('aria-checked', isBioActive.toString());
 if (isBioActive) {
 biometricsBtn.classList.remove('bg-surface-container-highest');
 biometricsBtn.classList.add('bg-secondary');
 knob.classList.remove('translate-x-1');
 knob.classList.add('translate-x-6');
 } else {
 biometricsBtn.classList.remove('bg-secondary');
 biometricsBtn.classList.add('bg-surface-container-highest');
 knob.classList.remove('translate-x-6');
 knob.classList.add('translate-x-1');
 }
 });
 }

 const submitBtn = document.getElementById('submit-btn');
 const termsCheck = document.getElementById('terms-check');
 const insuranceCheck = document.getElementById('insurance-check');

 function updateSubmitState() {
 if (!termsCheck || !insuranceCheck || !submitBtn) return;
 const isValid = termsCheck.checked && insuranceCheck.checked;
 if (isValid) {
 submitBtn.removeAttribute('disabled');
 submitBtn.classList.remove('opacity-50', 'pointer-events-none');
 } else {
 submitBtn.setAttribute('disabled', 'true');
 submitBtn.classList.add('opacity-50', 'pointer-events-none');
 }
 }

 if (termsCheck) termsCheck.addEventListener('change', updateSubmitState);
 if (insuranceCheck) insuranceCheck.addEventListener('change', updateSubmitState);
 })();
</script>
