<script>
 function switchTripState(state) {
 const tab1 = document.getElementById('tab-state-1');
 const tab2 = document.getElementById('tab-state-2');
 const tab3 = document.getElementById('tab-state-3');

 const statusHeading = document.getElementById('status-heading');
 const statusDesc = document.getElementById('status-description');
 const statusIcon = document.getElementById('status-icon');
 const etaPrimary = document.getElementById('eta-primary-text');
 const etaSub = document.getElementById('eta-sub-text');
 const pinCard = document.getElementById('pin-card');
 const carGroup = document.getElementById('car-marker-group');
 const pickupPill = document.getElementById('pickup-pill-text');

 const stepArriving = document.getElementById('step-arriving');
 const stepIntrip = document.getElementById('step-intrip');
 const stepDot4 = document.getElementById('step-dot-4');
 const stepLabel4 = document.getElementById('step-label-4');
 const stepLine3 = document.getElementById('step-line-3');

 // Reset tabs styling
 [tab1, tab2, tab3].forEach(t => {
 t.className = 'flex-1 py-2 px-1 rounded-lg text-center transition-all text-on-surface-variant hover:text-on-surface';
 });

 if (state === 1) {
 // State 1: Driver on the way
 tab1.className = 'flex-1 py-2 px-1 rounded-lg text-center transition-all bg-primary-container text-surface shadow-sm';
 statusHeading.textContent = 'Erik is on the way';
 statusDesc.textContent = 'Approaching from Dronning Eufemias gate';
 statusIcon.textContent = 'navigation';
 etaPrimary.textContent = 'ETA 3 min';
 etaSub.textContent = '· 1.2 km';
 pickupPill.textContent = 'Karl Johans gate 22';
 pinCard.classList.remove('hidden');

 // Map marker state
 carGroup.setAttribute('transform', 'translate(140, 250) rotate(-38)');

 // Stepper
 stepArriving.classList.remove('opacity-60');
 stepIntrip.classList.add('opacity-60');
 stepDot4.className = 'w-2 h-2 rounded-full bg-outline transition-colors duration-300';
 stepLabel4.className = 'font-label-sm text-label-sm text-on-surface-variant';
 stepLine3.className = 'h-0.5 w-4 bg-outline-variant transition-colors duration-300';
 } else if (state === 2) {
 // State 2: Driver arrived
 tab2.className = 'flex-1 py-2 px-1 rounded-lg text-center transition-all bg-primary-container text-surface shadow-sm';
 statusHeading.textContent = 'Driver arrived';
 statusDesc.textContent = 'Polestar 2 parked at designated pickup spot';
 statusIcon.textContent = 'location_on';
 etaPrimary.textContent = 'Arrived';
 etaSub.textContent = '· Waiting outside';
 pickupPill.textContent = 'Karl Johans gate 22';
 pinCard.classList.remove('hidden');

 // Map marker arrived at pickup point
 carGroup.setAttribute('transform', 'translate(55, 315) rotate(-15)');

 // Stepper
 stepArriving.classList.remove('opacity-60');
 stepIntrip.classList.add('opacity-60');
 stepDot4.className = 'w-2 h-2 rounded-full bg-outline transition-colors duration-300';
 stepLabel4.className = 'font-label-sm text-label-sm text-on-surface-variant';
 stepLine3.className = 'h-0.5 w-4 bg-outline-variant transition-colors duration-300';
 } else if (state === 3) {
 // State 3: In trip
 tab3.className = 'flex-1 py-2 px-1 rounded-lg text-center transition-all bg-primary-container text-surface shadow-sm';
 statusHeading.textContent = 'Heading to Oslo Lufthavn';
 statusDesc.textContent = 'Optimal route via E6 · Smooth traffic';
 statusIcon.textContent = 'flight_takeoff';
 etaPrimary.textContent = 'ETA 34 min';
 etaSub.textContent = '· 47.8 km';
 pickupPill.textContent = 'Destination: OSL Terminal 1';
 pinCard.classList.add('hidden'); // Hide entry pin during actual transit

 // Map marker moving forward along route
 carGroup.setAttribute('transform', 'translate(260, 150) rotate(-42)');

 // Stepper update
 stepArriving.classList.add('opacity-60');
 stepIntrip.classList.remove('opacity-60');
 stepDot4.className = 'w-2.5 h-2.5 rounded-full bg-secondary transition-colors duration-300';
 stepLabel4.className = 'font-label-sm text-label-sm text-secondary font-semibold';
 stepLine3.className = 'h-0.5 w-4 bg-secondary transition-colors duration-300';
 }
 }

 function toggleTrafficPill(btn) {
 btn.classList.toggle('bg-secondary-container');
 }

 function recenterMapView() {
 const carGroup = document.getElementById('car-marker-group');
 carGroup.classList.add('scale-125');
 setTimeout(() => {
 carGroup.classList.remove('scale-125');
 }, 350);
 }
