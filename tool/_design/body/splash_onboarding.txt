<script>
 (function() {
 let currentSlide = 0;
 const totalSlides = 3;
 const track = document.getElementById('slides-track');
 const pills = document.querySelectorAll('.indicator-pill');
 const nextBtn = document.getElementById('next-btn');
 const btnLabel = document.getElementById('btn-label');
 const btnIcon = document.getElementById('btn-icon');
 const skipBtn = document.getElementById('skip-btn');

 function updateCarousel(index) {
 currentSlide = Math.max(0, Math.min(index, totalSlides - 1));
 track.style.transform = `translateX(-${currentSlide * 100}%)`;

 pills.forEach((pill, i) => {
 if (i === currentSlide) {
 pill.classList.remove('w-2', 'bg-outline-variant');
 pill.classList.add('w-8', 'bg-primary');
 } else {
 pill.classList.remove('w-8', 'bg-primary');
 pill.classList.add('w-2', 'bg-outline-variant');
 }
 });

 if (currentSlide === totalSlides - 1) {
 btnLabel.textContent = 'Get Started';
 btnIcon.textContent = 'check';
 nextBtn.classList.remove('bg-secondary');
 nextBtn.classList.add('bg-primary-container');
 } else {
 btnLabel.textContent = 'Continue';
 btnIcon.textContent = 'arrow_forward';
 nextBtn.classList.add('bg-secondary');
 nextBtn.classList.remove('bg-primary-container');
 }
 }

 nextBtn.addEventListener('click', () => {
 if (currentSlide < totalSlides - 1) {
 updateCarousel(currentSlide + 1);
 } else {
 updateCarousel(0);
 }
 });

 skipBtn.addEventListener('click', () => {
 updateCarousel(totalSlides - 1);
 });

 pills.forEach((pill) => {
 pill.addEventListener('click', () => {
 const index = parseInt(pill.getAttribute('data-index'), 10);
 updateCarousel(index);
 });
 });

 // Touch Swipe Support
 let startX = 0;
 let endX = 0;

 track.parentElement.addEventListener('touchstart', (e) => {
 startX = e.touches[0].clientX;
 }, { passive: true });

 track.parentElement.addEventListener('touchend', (e) => {
 endX = e.changedTouches[0].clientX;
 const diff = startX - endX;
 if (Math.abs(diff) > 40) {
 if (diff > 0 && currentSlide < totalSlides - 1) {
 updateCarousel(currentSlide + 1);
 } else if (diff < 0 && currentSlide > 0) {
 updateCarousel(currentSlide - 1);
 }
 }
 }, { passive: true });
 })();
</script>
