<script>
 function toggleDetails() {
 const drawer = document.getElementById('receipt-drawer');
 const arrow = document.getElementById('receipt-arrow');
 const isHidden = drawer.classList.contains('hidden');
 
 if (isHidden) {
 drawer.classList.remove('hidden');
 drawer.classList.add('flex');
 arrow.style.transform = 'rotate(180deg)';
 } else {
 drawer.classList.add('hidden');
 drawer.classList.remove('flex');
 arrow.style.transform = 'rotate(0deg)';
 }
 }

 function switchFilter(tab) {
 const pastBtn = document.getElementById('tab-past');
 const scheduledBtn = document.getElementById('tab-scheduled');
 const pastContent = document.getElementById('past-trips-content');
 const scheduledContent = document.getElementById('scheduled-trips-content');

 if (tab === 'past') {
 pastBtn.className = 'flex-1 py-2.5 rounded-lg font-label-lg text-label-lg text-on-primary bg-primary transition-all duration-200 shadow-sm flex items-center justify-center gap-1.5';
 scheduledBtn.className = 'flex-1 py-2.5 rounded-lg font-label-lg text-label-lg text-on-surface-variant transition-all duration-200 flex items-center justify-center gap-1.5';
 pastContent.classList.remove('hidden');
 pastContent.classList.add('flex');
 scheduledContent.classList.add('hidden');
 scheduledContent.classList.remove('flex');
 } else {
 scheduledBtn.className = 'flex-1 py-2.5 rounded-lg font-label-lg text-label-lg text-on-primary bg-primary transition-all duration-200 shadow-sm flex items-center justify-center gap-1.5';
 pastBtn.className = 'flex-1 py-2.5 rounded-lg font-label-lg text-label-lg text-on-surface-variant transition-all duration-200 flex items-center justify-center gap-1.5';
 scheduledContent.classList.remove('hidden');
 scheduledContent.classList.add('flex');
 pastContent.classList.add('hidden');
 pastContent.classList.remove('flex');
 }
 }

 function showToast(message) {
 const toast = document.getElementById('toast');
 const text = document.getElementById('toast-text');
 text.textContent = message;
 toast.classList.remove('opacity-0', 'pointer-events-none');
 toast.classList.add('opacity-100');
 setTimeout(() => {
 toast.classList.add('opacity-0', 'pointer-events-none');
 toast.classList.remove('opacity-100');
 }, 2800);
 }

 function downloadReceiptNotice() {
 showToast('PDF Receipt sent to your BankID inbox');
 }

 function rebookTrip(route) {
 showToast('Routing to booking: ' + route);
 }
</script>
