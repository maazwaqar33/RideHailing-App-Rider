<script>
 // Password Visibility Toggle
 const togglePassBtn = document.getElementById('togglePassword');
 const passwordInput = document.getElementById('passwordInput');
 const eyeIcon = document.getElementById('eyeIcon');

 if (togglePassBtn && passwordInput && eyeIcon) {
 togglePassBtn.addEventListener('click', () => {
 const isPassword = passwordInput.getAttribute('type') === 'password';
 passwordInput.setAttribute('type', isPassword ? 'text' : 'password');
 eyeIcon.textContent = isPassword ? 'visibility_off' : 'visibility';
 });
 }

 // Language Switcher Toggle
 const langToggle = document.getElementById('langToggle');
 const langLabel = document.getElementById('langLabel');
 let isNorwegian = true;

 if (langToggle && langLabel) {
 langToggle.addEventListener('click', () => {
 isNorwegian = !isNorwegian;
 langLabel.textContent = isNorwegian ? 'NO (Norsk)' : 'EN (English)';
 });
 }

 // Tabs switcher micro-interaction
 const tabLogin = document.getElementById('tabLogin');
 const tabSignup = document.getElementById('tabSignup');
 const submitBtn = document.getElementById('submitBtn');

 if (tabLogin && tabSignup && submitBtn) {
 tabLogin.addEventListener('click', () => {
 tabLogin.className = "flex-1 py-2.5 rounded-lg font-label-lg text-label-lg transition-all duration-200 bg-surface-container-lowest text-primary shadow-sm font-semibold";
 tabSignup.className = "flex-1 py-2.5 rounded-lg font-label-lg text-label-lg transition-all duration-200 text-on-surface-variant hover:text-primary";
 submitBtn.querySelector('span:first-child').textContent = "Log In";
 });

 tabSignup.addEventListener('click', () => {
 tabSignup.className = "flex-1 py-2.5 rounded-lg font-label-lg text-label-lg transition-all duration-200 bg-surface-container-lowest text-primary shadow-sm font-semibold";
 tabLogin.className = "flex-1 py-2.5 rounded-lg font-label-lg text-label-lg transition-all duration-200 text-on-surface-variant hover:text-primary";
 submitBtn.querySelector('span:first-child').textContent = "Create Account";
 });
 }

 // Biometrics Delight Trigger
 const bioBtn = document.getElementById('bioBtn');
 if (bioBtn) {
 bioBtn.addEventListener('click', () => {
 bioBtn.classList.add('scale-90');
 setTimeout(() => {
 bioBtn.classList.remove('scale-90');
 alert('Biometric verification recognized for Oslo Rider pass.');
 }, 180);
 });
 }
</script>
