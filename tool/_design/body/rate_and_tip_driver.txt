<script>
 // Micro-interactions for Star Rating
 const starButtons = document.querySelectorAll('#star-rating-container .star-btn');
 let currentRating = 5;

 starButtons.forEach(btn => {
 btn.addEventListener('click', () => {
 const val = parseInt(btn.getAttribute('data-value'), 10);
 currentRating = val;
 starButtons.forEach((sBtn, idx) => {
 const starIcon = sBtn.querySelector('.material-symbols-outlined');
 if (idx < val) {
 sBtn.classList.remove('text-outline-variant');
 sBtn.classList.add('text-tertiary-fixed-dim');
 starIcon.style.fontVariationSettings = "'FILL' 1";
 } else {
 sBtn.classList.remove('text-tertiary-fixed-dim');
 sBtn.classList.add('text-outline-variant');
 starIcon.style.fontVariationSettings = "'FILL' 0";
 }
 });
 });
 });

 // Micro-interactions for Feedback Tags
 const tagButtons = document.querySelectorAll('#feedback-tags .tag-pill');
 tagButtons.forEach(tag => {
 tag.addEventListener('click', () => {
 const isSelected = tag.getAttribute('data-selected') === 'true';
 if (isSelected) {
 tag.setAttribute('data-selected', 'false');
 tag.className = 'tag-pill px-3 py-2 rounded-full font-label-md text-label-md transition-colors bg-surface-container text-on-surface hover:bg-surface-container-high';
 } else {
 tag.setAttribute('data-selected', 'true');
 tag.className = 'tag-pill px-3 py-2 rounded-full font-label-md text-label-md transition-colors bg-primary text-on-primary shadow-xs';
 }
 });
 });

 // Micro-interactions for Tip Buttons
 const tipButtons = document.querySelectorAll('#tip-options-row .tip-btn');
 tipButtons.forEach(btn => {
 btn.addEventListener('click', () => {
 tipButtons.forEach(other => {
 other.removeAttribute('data-selected');
 other.className = 'tip-btn h-14 rounded-lg flex flex-col items-center justify-center font-label-md text-label-md transition-all bg-surface-container text-on-surface hover:bg-surface-container-high';
 });
 btn.setAttribute('data-selected', 'true');
 btn.className = 'tip-btn h-14 rounded-lg flex flex-col items-center justify-center font-label-md text-label-md transition-all bg-secondary text-on-secondary shadow-sm scale-102 font-semibold';
 });
 });

 // Simple feedback on submission click
 const submitBtn = document.getElementById('submit-btn');
 submitBtn.addEventListener('click', () => {
 submitBtn.innerHTML = '<span class="material-symbols-outlined animate-spin text-[20px]">progress_activity</span>
<span>Submitting...</span>';
 setTimeout(() => {
 submitBtn.innerHTML = '<span class="material-symbols-outlined text-[20px]">check_circle</span>
<span>Thank you, Erik notified!</span>';
 submitBtn.classList.remove('bg-primary-container');
 submitBtn.classList.add('bg-secondary');
 }, 700);
 });
</script>
