<script>
 function selectTier(element, tierName, price) {
 const cards = document.querySelectorAll('.vehicle-card');
 cards.forEach(card => {
 card.classList.remove('is-selected', 'bg-secondary-container/20', 'shadow-md');
 card.classList.add('bg-surface-container-lowest', 'shadow-sm');
 const indicator = card.querySelector('.selection-indicator');
 if (indicator) {
 indicator.classList.remove('bg-secondary', 'text-on-secondary');
 indicator.classList.add('bg-surface-container', 'text-transparent');
 }
 });

 element.classList.remove('bg-surface-container-lowest', 'shadow-sm');
 element.classList.add('is-selected', 'bg-secondary-container/20', 'shadow-md');
 
 const indicator = element.querySelector('.selection-indicator');
 if (indicator) {
 indicator.classList.remove('bg-surface-container', 'text-transparent');
 indicator.classList.add('bg-secondary', 'text-on-secondary');
 }

 const btnLabel = document.getElementById('button-label');
 if (btnLabel) {
 btnLabel.textContent = `Request ${tierName} · ${price}`;
 }
 }
</script>
